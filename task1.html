'========================================================= Private Function FindMatchingDeals( _ ByVal dealsByKey As Object, _ ByVal usedDeals As Object, _ ByVal matchKey As String, _ ByVal targetAmount As Double _ ) As Collection Dim result As New Collection Dim candidateDeals As Collection Dim availableDeals As New Collection Dim deal As Object Dim i As Long If targetAmount <= 0 Then Set FindMatchingDeals = result Exit Function End If If Not dealsByKey.Exists(matchKey) Then Set FindMatchingDeals = result Exit Function End If Set candidateDeals = dealsByKey(matchKey) For i = 1 To candidateDeals.Count Set deal = candidateDeals(i) If Not usedDeals.Exists(CStr(deal("RowNo"))) Then availableDeals.Add deal End If Next i 'Single exact amount match For i = 1 To availableDeals.Count Set deal = availableDeals(i) If AmountsEqual(ToNumber(deal("Amount")), targetAmount) Then result.Add deal Set FindMatchingDeals = result Exit Function End If Next i 'One-to-many match If availableDeals.Count > 0 Then Set result = FindSubsetByAmount(availableDeals, targetAmount) End If Set FindMatchingDeals = result End Function Private Function FindSubsetByAmount(ByVal deals As Collection, ByVal targetAmount As Double) As Collection Const MAX_CANDIDATES As Long = 40 ' cap on how many candidate deals feed the search Const MAX_COMBINATION_SIZE As Long = 4 ' cap on how many deals can be combined into one GL match Dim sortedDeals As Collection Dim result As New Collection Set sortedDeals = SortDealsByAmountDesc(deals) If sortedDeals.Count > MAX_CANDIDATES Then Set sortedDeals = FirstNDeals(sortedDeals, MAX_CANDIDATES) End If If TryCombination(sortedDeals, targetAmount, 1, 0, 0, MAX_COMBINATION_SIZE, result) Then Set FindSubsetByAmount = result Else Set FindSubsetByAmount = New Collection End If End Function ' Depth-capped combination search - the successor to both the original ' unbounded recursive backtracking (exponential, could hang on large deal ' groups) and an intermediate DP-based subset-sum (polynomial per call, but ' still expensive enough per GL row that a full multi-report run could add ' up to a very long time). ' ' Capping the combination size (MAX_COMBINATION_SIZE) bounds the search to ' at most SUM(C(n,1)..C(n,maxDepth)) node visits - a small, FIXED ceiling ' regardless of how many candidate deals exist in a bucket. In practice a ' GL balance being the sum of more than 3-4 separate deal tickets is very ' unusual, so this does not meaningfully change which real matches are ' found, but it guarantees the search can never run away. Private Function TryCombination( _ ByVal deals As Collection, _ ByVal targetAmount As Double, _ ByVal startIndex As Long, _ ByVal runningTotal As Double, _ ByVal depth As Long, _ ByVal maxDepth As Long, _ ByRef pickedDeals As Collection _ ) As Boolean Dim i As Long Dim deal As Object Dim nextTotal As Double If depth > 0 And AmountsEqual(runningTotal, targetAmount) Then TryCombination = True Exit Function End If If depth >= maxDepth Then TryCombination = False Exit Function End If For i = startIndex To deals.Count Set deal = deals(i) nextTotal = runningTotal + ToNumber(deal("Amount")) If nextTotal <= targetAmount + 0.01 Then pickedDeals.Add deal If TryCombination(deals, targetAmount, i + 1, nextTotal, depth + 1, maxDepth, pickedDeals) Then TryCombination = True Exit Function End If pickedDeals.Remove pickedDeals.Count End If Next i TryCombination = False End Function '========================================================= ' Output handling ' Full layout (HasCurrencyColumn = True): ' B Serial no. | C Client | D Country | E Region | F Transaction ID ' G Type of facility | H Currency | I Value Date | J Maturity Date ' K Maturity Remaining | L Interest Rate | M Principal | N USD Equiv. Private Function ColLetter(ByVal colNum As Long) As String Dim n As Long Dim rem_ As Long Dim s As String n = colNum Do While n > 0 rem_ = (n - 1) Mod 26 s = Chr(65 + rem_) & s n = (n - rem_ - 1) \ 26 Loop ColLetter = s End Function
